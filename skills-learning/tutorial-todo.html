# 待办事项 Skill

## When to Activate
- 用户说"待办"、"我的任务"
- 用户说"记一下..."添加任务
- 用户说"完成了..."标记完成
- 用户询问"有什么待办"

## Actions

### add - 添加待办
"帮我记一下，明天要交报告"
→ 创建新待办，保存到本地

### list - 查看列表
"我的待办"
→ 按优先级显示所有待办

### complete - 完成
"完成了第一个" 或 "完成了项目方案"
→ 标记为已完成

### delete - 删除
"删除第二个" 或 "删掉那个报告"
→ 删除待办

### clear - 清除已完成
"清除已完成"
→ 删除所有已完成项目

/**
 * 添加待办
 */
async function addTodo(content, options = {}) {
  const todo = {
    id: crypto.randomUUID(),
    content,
    priority: options.priority || 'medium',
    dueDate: options.dueDate || null,
    completed: false,
    createdAt: Date.now(),
    completedAt: null
  };

  const todos = await getTodos();
  todos.push(todo);
  await saveTodos(todos);

  return formatTodoAdded(todo);
}

/**
 * 列出待办
 */
async function listTodos() {
  const todos = await getTodos();
  const activeTodos = todos.filter(t => !t.completed);

  if (activeTodos.length === 0) {
    return '🎉 太棒了！没有待办事项了';
  }

  // 按优先级分组
  const byPriority = {
    high: activeTodos.filter(t => t.priority === 'high'),
    medium: activeTodos.filter(t => t.priority === 'medium'),
    low: activeTodos.filter(t => t.priority === 'low')
  };

  let output = '📋 待办事项\n━━━━━━━━━━━━━━━━━\n';

  if (byPriority.high.length) {
    output += `\n🔴 高优先级 (\${byPriority.high.length})\n`;
    byPriority.high.forEach((t, i) => {
      output += \`  \${i + 1}. \${t.content}\`;
      if (t.dueDate) output += \` - \${formatDate(t.dueDate)}\`;
      output += '\n';
    });
  }

  // ... 中低优先级同理

  return output;
}

/**
 * 完成待办
 */
async function completeTodo(identifier) {
  const todos = await getTodos();
  const index = findTodoIndex(todos, identifier);

  if (index === -1) {
    throw new Error('找不到这个待办');
  }

  todos[index].completed = true;
  todos[index].completedAt = Date.now();
  await saveTodos(todos);

  return \`✅ 完成！"\${todos[index].content}" 已标记完成\`;
}

🏷️ 优先级识别

从自然语言中识别优先级：

/**
 * 从文本中识别优先级
 */
function extractPriority(text) {
  const lower = text.toLowerCase();

  if (/紧急|紧急|马上|立刻|asap/.test(lower)) {
    return 'high';
  }

  if (/重要|必须|尽快/.test(lower)) {
    return 'medium';
  }

  return 'low';
}

/**
 * 从文本中识别截止日期
 */
function extractDueDate(text) {
  const now = new Date();

  // 今天/明天
  if (/今天/.test(text)) {
    return now;
  }

  if (/明天/.test(text)) {
    const tomorrow = new Date(now);
    tomorrow.setDate(tomorrow.getDate() + 1);
    return tomorrow;
  }

  // 本周/下周
  if (/本周/.test(text)) {
    const friday = new Date(now);
    while (friday.getDay() !== 5) friday.setDate(friday.getDate() + 1);
    return friday;
  }

  // 特定日期 "4月30日"
  const dateMatch = text.match(/(\d+)月(\d+)日/);
  if (dateMatch) {
    const date = new Date();
    date.setMonth(parseInt(dateMatch[1]) - 1);
    date.setDate(parseInt(dateMatch[2]));
    return date;
  }

  return null;
}
