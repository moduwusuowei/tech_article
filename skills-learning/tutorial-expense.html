- 用户说"花了XX"、"消费了XX"

- 用户说"设置预算"

/**
 * 解析自然语言输入
 */
function parseTransaction(input) {
  // 提取金额
  const amountMatch = input.match(/(\d+(?:\.\d+)?)\s*(?:块|元)?/);
  const amount = amountMatch ? parseFloat(amountMatch[1]) : 0;

  // 识别收入/支出
  const type = /入账|收入|到账|发工资/.test(input) ? 'income' : 'expense';

  // 识别分类
  const category = classifyTransaction(input, type);

  return { amount, type, category, note: input };
}

/**
 * 分类识别
 */
function classifyTransaction(input, type) {
  const categories = CATEGORIES[type];

  for (const cat of categories) {
    for (const pattern of cat.patterns) {
      if (input.includes(pattern)) {
        return cat.name;
      }
    }
  }

  return '其他';
}

/**
 * 月度消费报告
 */
async function getMonthlyReport(year, month) {
  const transactions = await getTransactions();
  const startDate = new Date(year, month - 1, 1);
  const endDate = new Date(year, month, 0);

  // 筛选当月交易
  const monthlyTxns = transactions.filter(t =>
    t.date >= startDate && t.date <= endDate
  );

  // 按分类统计
  const byCategory = {};
  let total = 0;

  for (const txn of monthlyTxns) {
    if (txn.type === 'expense') {
      byCategory[txn.category] = (byCategory[txn.category] || 0) + txn.amount;
      total += txn.amount;
    }
  }

  // 生成报告
  let report = '📊 ' + formatYearMonth(year, month) + '消费报告\n';
  report += '━━━━━━━━━━━━━━━━━\n';
  report += '💸 总支出：' + formatAmount(total) + ' 元\n';
  report += '📈 日均：' + formatAmount(total / endDate.getDate()) + ' 元\n';
  report += '━━━━━━━━━━━━━━━━━\n';

  // 分类详情
  const sorted = Object.entries(byCategory)
    .sort((a, b) => b[1] - a[1]);

  for (const [cat, amount] of sorted) {
    const pct = ((amount / total) * 100).toFixed(0);
    const bar = '█'.repeat(Math.floor(pct / 5));
    report += cat.padEnd(4) + ' ' + formatAmount(amount).padStart(7);
    report += ' (' + pct + '%) ' + bar + '\n';
  }

  return report;
}

/**
 * 预算超支检查

async function checkBudgetAlert(total, category) {
  const budgets = await getBudgets();
  const budget = budgets[category];

  if (!budget) return null;

  const spent = await getCategoryTotal(category);
  const pct = (spent / budget * 100).toFixed(0);

  if (spent > budget) {
    return '⚠️ ' + category + '超支 ' + pct + '%';
  } else if (spent > budget * 0.8) {
    return '💡 ' + category + '已使用 ' + pct + '%';
  }

  return null;
}

💾 IndexedDB 存储

使用 IndexedDB 实现数据持久化：

/**
 * IndexedDB 数据库操作
 */
const DB_NAME = 'expense_tracker';
const DB_VERSION = 1;
const STORE_NAME = 'transactions';

let db = null;

/**
 * 打开数据库
 */
async function openDB() {
  return new Promise((resolve, reject) => {
    const request = indexedDB.open(DB_NAME, DB_VERSION);

    request.onerror = () => reject(request.error);
    request.onsuccess = () => {
      db = request.result;
      resolve(db);
    };

    request.onupgradeneeded = (event) => {
      const database = event.target.result;
      if (!database.objectStoreNames.contains(STORE_NAME)) {
        const store = database.createObjectStore(STORE_NAME, { keyPath: 'id' });
        store.createIndex('date', 'date');
        store.createIndex('category', 'category');
        store.createIndex('type', 'type');
      }
    };
  });
}

/**
 * 保存交易记录
 */
async function saveTransactions(transactions) {
  const database = await openDB();
  const tx = database.transaction(STORE_NAME, 'readwrite');
  const store = tx.objectStore(STORE_NAME);

  // 清除旧数据
  store.clear();

  // 添加新数据
  for (const t of transactions) {
    store.add(t);
  }

  return new Promise((resolve, reject) => {
    tx.oncomplete = () => resolve();
    tx.onerror = () => reject(tx.error);
  });
}

/**
 * 获取所有交易
 */
async function getTransactions() {
  const database = await openDB();
  const tx = database.transaction(STORE_NAME, 'readonly');
  const store = tx.objectStore(STORE_NAME);

  return new Promise((resolve, reject) => {
    const request = store.getAll();
    request.onsuccess = () => resolve(request.result);
    request.onerror = () => reject(request.error);
  });
}
